class Model_Settings extends Model
{
    public function get($key)
    {
        $sql = "SELECT setting_value
                FROM settings
                WHERE setting_key = :key
                LIMIT 1";

        $stmt = self::$conn->prepare($sql);
        $stmt->execute([
            ':key' => $key
        ]);

        return $stmt->fetchColumn();
    }
}
